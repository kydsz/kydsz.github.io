> 降序排序后三重循环枚举三元组，当前最大值不足时剪枝求最大和。

## 暴力

```c++
#include <bits/stdc++.h>
using namespace std;
int a[10000000];
int main() {
    int n, k, ans = 0, sum;;
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n, [](int a1, int a2) { return a1 > a2; });
    for (int i = 0; i < n - 2; ++i) {
        for (int j = i + 1; j < n - 1; ++j) {
            for (int l = j + 1; l < n; ++l) {
                sum = a[i] + a[j] + a[l];
                if (sum < ans) break;
                if (sum % k == 0) ans = sum;
            }
            if (a[i] + a[j] + a[j + 1] < ans) break;
        }
        if (a[i] + a[i + 1] + a[i + 2] < ans) break;
    }
    cout << ans << endl;
    return 0;
}
```
