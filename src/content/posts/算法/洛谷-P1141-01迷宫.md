> BFS 求连通块：只能走向数字与当前格相反的相邻格，一次 BFS 标记整个连通块并记录大小，块内各点答案相同。

# 广度优先算法BFS

```c++
#include<iostream>
#include<queue>
#define MAX 1010
using namespace std;

typedef struct _Node{
    int x, y;
}Node;
queue<Node> qu;
int map[MAX][MAX], state[MAX][MAX];
int result[MAX*MAX];  // 能联通的坐标结果都一样

//移动
int dx[4] = {1, 0, -1, 0};
int dy[4] = {0, 1, 0, -1};
int n, m, flag = 0;


int bfs(int x, int y) {
    int step = 1;
    flag++;
    state[x][y] = flag;
    Node temp;
    temp.x = x;
    temp.y = y;
    qu.push(temp);
    while (!qu.empty()){
        for (int i = 0; i < 4; ++i) {
            int tx = qu.front().x + dx[i];
            int ty = qu.front().y + dy[i];

            if (tx < 0 || ty < 0 || tx >= n || ty >= n || state[tx][ty] != 0) continue;
            if ((map[tx][ty] == 0 && map[qu.front().x][qu.front().y] == 1) || map[tx][ty] == 1 && map[qu.front().x][qu.front().y] == 0){
                state[tx][ty] = flag;
                temp.x = tx;
                temp.y = ty;
                qu.push(temp);
                step++;
            }
        }
        qu.pop();
    }
    result[flag] = step;
    return step;
}


int main(){
    cin >> n >> m;
    string s;
    for (int i = 0; i < n; ++i) {
        cin >> s;
        for (int j = 0; j < n; ++j) {
            map[i][j] = s[j]-'0';
        }
    }
    

    int x, y;
    while (m--){
        cin >> x >> y;
        x--, y--;
        if (state[x][y] != 0)
            cout << result[state[x][y]] << endl;
        else
            cout << bfs(x, y) << endl;
    }
    return 0;
}
```
