> KMP：next 数组记录最长相等前后缀，失配时回退避免重复匹配。

## kmp

```c++
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;


int kmpNext[10000000];
vector<int> arr;

void getNext(const string &b){
    kmpNext[0] = 0;
    for (int i = 1, j = 0; i < b.length(); ++i) {
        while (j > 0 && b[i] != b[j]) j = kmpNext[j - 1];
        
        if (b[i] == b[j]) j++;
        kmpNext[i] = j;
    }
}

void kmp(const string &a, const string &b){
    getNext(b);
    for (int i = 0, j = 0; i < a.length(); ++i) {
        while (j > 0 && a[i] != b[j]) j = kmpNext[j - 1];
        
        if (a[i] == b[j]) j++;
        if (j == b.length()) {
            arr.push_back(i - b.length() + 1);
        }
    }
}


int main() {
    string s, t;
    cin >> s >> t;
    kmp(s, t);
    for (int x: arr) {
        cout << x + 1 << endl;
    }
    for (int i = 0; i < t.length(); ++i) {
        cout << kmpNext[i] << ' ';
    }
    return 0;
}
```
