> 反向建边后按编号从大到小 DFS，每个点首次访问即记录最大可达编号。

## 反向建边 + dfs

```c++
#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int n, m, A[N];
vector<int> G[N];

void dfs(int x, int w){
    if (A[x]) return;
    A[x] = w;
    for (auto item: G[x]) {
        dfs(item, w);
    }
}


int main() {
    // 不关闭同步只能60分
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m;
    int u, v;
    for (int i = 0; i < m; ++i) {
        cin >> u >> v;
        G[v].push_back(u);
    }
    for (int i = n; i > 0; --i) {
        dfs(i,i);
    }
    for (int i = 1; i <= n; ++i) {
        cout << A[i] << ' ';
    }

    return 0;
}
```

## 链式前向星60分

```c++
#include <iostream>
#include <cstring>
#include <vector>

using namespace std;
int n, m;
const int N = 1e5 + 10;
struct Edge {
    int to, next;
} edge[N];
int head[N], cut;

void init() {
    memset(head, -1, sizeof head);
    cut = 0;
}

void addE(int u, int v) {
    edge[cut].to = v;
    edge[cut].next = head[u];
    head[u] = cut++;
}

vector<bool> state(N);

int dfs(int index) {
    int maxE = -0xfff;
    state[index] = true;
    maxE = max(maxE, index);
    if (head[index] == -1) return index;
    for (int i = head[index]; i != -1; i = edge[i].next) {
        if (!state[edge[i].to]) {
            maxE = max(maxE, dfs(edge[i].to));
        }

    }
    return maxE;
}


int main() {
    init();
    cin >> n >> m;
    int u, v;
    for (int i = 0; i < m; ++i) {
        cin >> u >> v;
        addE(u, v);
    }
    for (int i = 1; i <= n; i++) {
        fill(state.begin(), state.end(),0);
        cout << dfs(i) << ' ';
    }
    return 0;
}
```

## bfs

```c++
#include <iostream>
#include <queue>
#include <cstring>
using namespace std;
bool graph[10010][10010];

int bfs(int n, int u) {
    int ans = u;
    queue<int> q;
    int state[10010];
    memset(state, 0, sizeof state);
    q.push(u);
    while (!q.empty()) {
        int f = q.front();
        q.pop();
        for (int i = 1; i <= n; ++i) {
            if (graph[f][i] && !state[i]) {
                state[i] = true;
                q.push(i);
                ans = max(ans, i);
            }
        }
    }
    return ans;
}


int main() {
    int n, m, u, v;
    cin >> n >> m;
    while (m--) {
        cin >> u >> v;
        graph[u][v] = true;
    }
    for (int i = 1; i <= n; ++i) {
        cout << bfs(n, i) << ' ';
    }
    return 0;
}
```
