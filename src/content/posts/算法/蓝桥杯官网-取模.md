> 双重循环枚举 x<y，判断是否存在 n%x 等于 n%y。

```c++
#include <iostream>

using namespace std;


bool solve(int n, int m) {
    for (int y = 1; y <= m; y++) {
        for (int x = 1; x < y; x++) {
            if (n % x == n % y) return true;
        }
    }
    return false;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        if (solve(n, m)) {
            cout << "Yes" << endl;
        } else {
            cout << "No" << endl;
        }
    }
    return 0;
}
```
