> 差分数组将区间加转为两端点修改，前缀和还原后求最小值。

# 差分

```c++
#include<iostream>
#include <algorithm>

using namespace std;
int d[5000001];
int a[5000001];

int main(){
    int n, p;
    cin >> n >> p;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
        d[i] = a[i] - a[i-1];
    }

    int x, y, z;
    for (int i = 0; i < p; ++i) {
        cin >> x >> y >> z;
        d[x] += z;
        d[y+1] -= z;
    }
    for (int i = 1; i <= n; ++i) {
        a[i] = d[i] + a[i-1];
    }

    int* path = min_element(&a[1], &a[n+1]);
    cout << *path;
    return 0;
}
```
