> 埃氏筛：未标记的数为素数，将其所有倍数标记为合数并计数。

# 质数筛

```c++
#include <iostream>

using namespace std;

long long n;
bool check[100000005];


int main() {
    cin >> n;
    long ans = 0;
    check[1] = true;
    for (long long i = 2; i <= n; ++i) {
        if (!check[i]) {
            ans++;
            for (long long j = 2 * i; j <= n; j += i) {
                check[j] = true;
            }
        }
    }
    cout << ans;
    return 0;
}
```
