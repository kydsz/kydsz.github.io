> DFS 回溯枚举组合：固定起始下标避免重复，取满 k 个数求和，再用试除法判断素数统计答案。

## dfs搜索



### 代码一

``` c++
#include <iostream>
#include <vector>

using namespace std;

void dfs(const vector<int> &arr, int k, int start, vector<int> &current, vector<vector<int>> &a) {
    // 当前组合的长度符合要求
    if (current.size() == k) {
        a.push_back(current);
        return;
    }

    // 从` start `开始迭代数组以确保组合是唯一的
    for (int i = start; i < arr.size(); ++i) {
        // 将当前元素添加到组合中
        current.push_back(arr[i]);
        // 对下一个元素进行递归
        dfs(arr, k, i + 1, current, a);
        // 删除要回溯的最后一个元素
        current.pop_back();
    }
}

bool isprime(int a) {
    for (int i = 2; i * i <= a; ++i) {
        if (a % i == 0) return false;
    }
    return true;
}

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> arr(n);
    for (int i = 0; i < n; ++i) {
        cin >> arr[i];
    }

    vector<int> current;
    vector<vector<int>> a;

    // 生成组合
    dfs(arr, k, 0, current, a);
    int ans = 0;
    // 打印所有组合
    for (const auto &han: a) {
        int sum = 0;
        for (int num: han) {
            sum += num;
        }
        if (isprime(sum)) ans++;
    }
    cout << ans;
    return 0;
}

```

```c++
#include <iostream>


using namespace std;

bool isprime(int a) {
    for (int i = 2; i * i <= a; i++)
        if (a % i == 0)
            return false;
    return true;
}
int n, k, ans;
int a[25];
void dfs(int m, int sum, int start) {
    if (m == k) {
        if (isprime(sum))
            ans++;
        return;
    }
    for (int i = start; i < n; i++) {
        dfs(m + 1, sum + a[i], i + 1);//递归
    }
}

int main() {
    cin >> n >> k;
    for (int i = 0; i < n; i++)
        cin >> a[i];
    dfs(0, 0, 0);
    cout << ans;
    return 0;
}
```
