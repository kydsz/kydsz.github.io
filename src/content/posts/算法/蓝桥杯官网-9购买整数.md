> 二分答案：a·n+b·位数随 n 单调递减，在区间上二分求最大可行 n。

## 二分

```c++
#include <iostream>

using namespace std;

long long a, b, x;

long long d(long long n) {
    long long len = 0;
    while (n) {
        len++;
        n /= 10;
    }
    return len;
}

bool check(long long n) {
    if (a * n + b * d(n) > x) {
        return false;
    }
    return true;
}

int main() {
    cin >> a >> b >> x;
    long long l = 1, r = 1e9 + 1, mid;
    while (l < r) {
        mid = (l + r) / 2;
        if (check(mid)) {
            l = mid + 1;
        } else {
            r = mid;
        }
    }
    cout << l - 1;

    return 0;
}
```
