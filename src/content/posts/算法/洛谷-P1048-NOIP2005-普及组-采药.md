> 01 背包：dp[j] 表示容量 j 下的最大价值，对每个草药倒序枚举容量，决策采（耗时 t、加价值 v）或不采。

## 动态规划	01背包

```c++
#include <bits/stdc++.h>
using namespace std;
const int N = 1050;
int n, m;
int t[N], v[N], dp[N];

int main() {
    cin >> m >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> t[i] >> v[i];
    }
    for (int i = 1; i <= n; ++i) {
        for (int j = m; j >= t[i]; --j) {
            dp[j] = max(dp[j], dp[j - t[i]] + v[i]);
        }
    }
    cout << dp[m];
    return 0;
}
```
