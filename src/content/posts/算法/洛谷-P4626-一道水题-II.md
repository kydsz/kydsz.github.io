> 求 1..n 的最小公倍数：欧拉筛出素数，累乘每个素数的最高次幂。

## 求n个数的最大公约数	欧拉筛	数学

####  求n个数的最大公约数就是这些数中每一个同样质因子的最小指数

##### 思路：求出n以内所有的质数后，枚举这些质数，求出一个最大数k使得prime的k次幂<n并把答案累乘模mod

```c++
#include <iostream>

using namespace std;
int prime[5761460];
bool vis[100000005];
long long ans = 1, mod = 100000007;
int n, cnt = 0;

void jisuan(int p) {
    long long total = 1;
    for (long long i = 1;; i++) {
        total *= p;
        if (total > n)
            break;
        ans *= p;
        ans %= mod;
    }
}

int main() {
    cin >> n;
    for (int i = 2; i <= n; i++) {
        if (!vis[i]) {
            prime[cnt++] = i;
        }
        for ( int j = 0; j < cnt && i * prime[j] <= n; j++) {
            vis[i * prime[j]] = true;
            if (i % prime[j] == 0) break;
        }
    }
    for ( int i = 0; i <= cnt - 1; i++) {
        if (prime[i] <= n) {
            jisuan(prime[i]);
        } else
            break;
    }
    cout << ans << endl;
    return 0;
}
```

例如：n = 10

小于10的质数为 2 3 5 7

ans = 2^3 + 3^2 + 5^1 +7^1 (2^3小于10，3^2小于10..)





### 80分 超内存

```c++
#include <bits/stdc++.h>

using namespace std;
const int mod = 100000007;
const int N = 1e8 + 10;
int n;
bool vis[N];
vector<int> prime;

void euler() {
    for (int i = 2; i <= n; ++i) {
        if (!vis[i]) {
            prime.push_back(i);
        }
        for (int j : prime) {
            if (i * j > n) break;
            vis[i * j] = true;
            if (i % j == 0) break;
        }
    }
}

long long power(int x){
    long long ans = 1;
    while (ans * x < n) {
        ans = ans * x;
    }
    return ans;
}

int main() {
    cin >> n;
    euler();
    long long ans = 1;
    for (auto x: prime) {
        ans = ans * power(x) % mod;
    }
    cout << ans;
    return 0;
}
```
