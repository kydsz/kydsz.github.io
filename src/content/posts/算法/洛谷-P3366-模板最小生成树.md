> Kruskal：边按权值升序排序，用并查集判环并累计边权。

# 最小生成树， krusal

``` c++
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int N = 210000;
long n, m, ans = 0, sum = 0;

struct _Edge{
    int u, v, w;
    bool operator<(const _Edge& other) const {
        return w < other.w;
    }
};

vector<_Edge> edges;
int F[N];

void initF(){
    for (int i = 0; i < n; ++i) {
        F[i] = i;
    }
}

int findF(int x){
    if (F[x] != x) return findF(F[x]);
    else return x;
}

void unionSet(int u, int v){
    int rootU = findF(u);
    int rootV = findF(v);
    if (rootU != rootV) {
        F[rootU] = rootV;
    }
}

bool checkCycle(const _Edge& e){
    int rootU = findF(e.u);
    int rootV = findF(e.v);
    return rootU == rootV;
}

void krus(){
    initF();
    sort(edges.begin(), edges.end());

    for (_Edge e : edges) {
        if (!checkCycle(e)) { // 防止形成环路
            unionSet(e.u, e.v);
            ans += e.w;
            sum++;
        }
    }
}

int main(){
    cin >> n >> m;
    edges.resize(m);
    for (int i = 0; i < m; ++i) {
        cin >> edges[i].u >> edges[i].v >> edges[i].w;
    }
    krus();
//    if (findF(1), findF(n))
//        cout << ans;
//    else
//        cout << "orz";
    if (sum + 1 == n) {
        cout << ans ;
    } else {
        cout << "orz";
    }
    return 0;
}
```
