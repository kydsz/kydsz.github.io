> f[i] 表示以 a[i] 结尾的最长上升子序列长度，枚举 j<i 且 a[j]<a[i] 时 f[i]=max(f[i],f[j]+1)，复杂度 O(n²)。

## 线性dp	lis

```c++
#include<iostream>

using namespace std;
const int N = 1e6 + 10;
int n;
int a[N];
int f[N];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    // dp[i]表示以第i项为结尾的最长上升子序列的长度
    for (int i = 1; i <= n; i++) {
        f[i] = 1;
        for (int j = 1; j < i; j++) {
            if (a[j] < a[i]) // 如果a[j] < a[i] 表示a[i]可以接在a[j]的后面
                f[i] = max(f[i], f[j] + 1);
        }
    }
    int res = 0;
    for (int i = 1; i <= n; i++)
        res = max(res, f[i]); // 求出字串中最长的
    cout << res << endl;
    return 0;
}
```
