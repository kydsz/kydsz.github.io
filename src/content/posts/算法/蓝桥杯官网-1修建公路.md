> Kruskal：边按长度升序，并查集判环合并，无法连通时输出 -1。

# 克鲁斯卡尔 	kruskal

### 最小生成树


``` c++
#include <bits/stdc++.h>
using namespace std;

const long long LEN = (long long)1e6+1;
long long n, m, ans = 0, SUM = 0;

struct _edge{
    long long a, b, l; // a到b的距离l
}edge[LEN];
long long f[LEN];

bool comp(struct _edge x, struct _edge y) {
    return x.l < y.l;
}

void initF(){
    
    for (long long i = 0; i < m; ++i) {
        f[i] = i;
    }
}

long long findF(long long x) {
    if (f[x] == x) return x;
    else return f[x] = findF(f[x]);
}

void kruskal() {
    sort(edge, edge + m, comp);
    initF();
    for (long long i = 0; i < m; ++i) {
        long long v = findF(edge[i].a);
        long long u = findF(edge[i].b);
        if (v == u) continue;
        ans += edge[i].l;
        SUM++;
        f[u] = v;
    }
}


int main() {
     cin >> n >> m; // n 小区数， m公路数
    for (long long i = 0; i < m; ++i) {
        cin >> edge[i].a >> edge[i].b >> edge[i].l;
    }
    
    kruskal();
    if (SUM + 1 == n) {
        cout << ans;
    } else cout << -1;
    return 0;
}
```
