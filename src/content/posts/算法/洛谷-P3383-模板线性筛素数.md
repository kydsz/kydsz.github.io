> 欧拉筛：每个合数只被最小质因子筛一次，当 i%prime[j]==0 即 break。

## 线性筛  欧拉筛
```c++
#include <iostream>

using namespace std;
typedef long long ll;
const int LEN = 100000010;

bool vis[LEN];
ll prime[LEN];
//欧拉筛函数

ll Euler_sieve(ll n) {
    ll i, j, k;
    k = 0;//保存素数的个数
    for (i = 2; i <= n; i++) {
        if (!vis[i])//i是素数，则存起来
            prime[k++] = i;
        for (j = 0; j < k; j++)//进行倍增，用以i去乘i之前(包括i)的素数
        {
            if (i * prime[j] > n)//倍增结果超出范围，退出
                break;

            vis[i * prime[j]] = true;//将倍增结果进行标记

            if (i % prime[j] == 0)//i是前面某个素数的倍数时，也需要退出 
                break;
        }
    }
    return k;
}

int main() {
    ios::sync_with_stdio(false);
    ll n, m;
    cin >> n >> m;
    Euler_sieve(n);
    while (m--) {
        ll q;
        cin >> q;
        cout << prime[q - 1] << endl;
    }
    return 0;
}
```

if(i%prime[j]==0)break;
prime数组中的素数是递增的,当 i 能整除 prime[j]，那么 i*prime[j+1] 这个合数肯定也可以被 prime[j] 筛掉，因为i中含有prime[j], prime[j] 比 prime[j+1] 小。
