> 树状数组：沿 lowbit 累加实现单点修改，前缀和作差求区间和。

## 树状数组

```c++
#include<iostream>

using namespace std;
const int TREE = 1000001;
const int ARR = 1000001;

int tree[TREE];
int arr[ARR];

int lowbit(int x) {
    return x & (-x);
}

void update(int i, int v) {
    while (i < TREE) {
        tree[i] += v;
        i += lowbit(i);
    }
}

long query(int i) {
    long sum = 0;
    while (i > 0) {
        sum += tree[i];
        i -= lowbit(i);
    }
    return sum;
}

int main() {
    int n, m; //数组长度n，以及查询操作次数m
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) {
        cin >> arr[i];
        update(i, arr[i]);
    }
    while (m--) {
        int x, y, k;
        cin >> x >> y >> k;

        if (x == 1) {
            update(y,k);
        } else{
            cout << query(k) - query(y- 1) << endl;
        }
    }
    return 0;
}
```
