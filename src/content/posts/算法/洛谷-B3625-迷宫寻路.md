> 从起点 DFS 四方向探索，遇墙或已访问即回溯，到达 (n,m) 时置标记提前返回。

# 深度优先算法DFS
```c++
#include <iostream>
using namespace std;

#define NUM 105

// n是迷宫的行数(y方向), m是迷宫的列数(x方向)
int n, m;
// map数组存储迷宫信息，res数组记录是否访问过
int map[NUM][NUM], res[NUM][NUM];
// judge标志是否找到从(1,1)到(n,m)的路径
bool judge = false;

// step_x和step_y数组定义了一组可能的移动方向
int step_x[4] = {-1, 0, 1, 0};  // 左右移动（x方向）
int step_y[4] = {0, -1, 0, 1};  // 上下移动（y方向）

// dfs函数实现深度优先搜索
void dfs(int y, int x){
    // 如果已经找到一条路径，或者(x,y)位置不在迷宫范围内，或者该位置是墙壁，或者已经访问过，则返回
    if (judge || y < 1 || x < 1 || y > n || x > m || map[y][x] == 1 || res[y][x] == 1) return;
    // 标记当前位置为已访问
    res[y][x] = 1;
    // 如果当前位置是出口，更新judge标志为true并返回
    if (y == n && x == m) {
        judge = true;
        return;
    }
    // 尝试向四个可能的方向移动
    for (int i = 0; i < 4; ++i) {
        dfs(y + step_y[i], x + step_x[i]);
    }
}

// main函数，程序的入口点
int main(){
    // 读入行数n和列数m
    cin >> n >> m;
    // 读入迷宫数据，'.'为路径，'#'为墙
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            char c;
            cin >> c;
            if (c == '#') {
                // '#'代表这是一面墙
                map[i][j] = 1;
            }
        }
    }
    // 从迷宫的起点(1,1)开始搜索路径
    dfs(1, 1);
    // 输出结果，如果judge为true，输出"Yes"；否则输出"No"
    cout << (judge ? "Yes" : "No");
    return 0;
}
```
