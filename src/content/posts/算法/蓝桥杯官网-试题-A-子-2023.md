> 四重循环数 '2023' 子序列个数，DP 用四状态分别累计 2、20、202、2023。

```c++
#include<bits/stdc++.h>

using namespace std;
typedef long long ll; // 开long long

string s;

ll f1() {
    ll cnt = 0;
    for (ll a = 0; a < s.size(); a++) {
        if (s[a] != '2') continue;
        for (ll b = a + 1; b < s.size(); b++) {
            if (s[b] != '0') continue;
            for (ll c = b + 1; c < s.size(); c++) {
                if (s[c] != '2') continue;
                for (ll d = c + 1; d < s.size(); d++) {
                    if (s[d] != '3') continue;
                    cnt++;
                }
            }
        }
    }
    return cnt;
}

ll f2() {
    ll dp[4] = {0,0,0,0}; // 分别代表"2"、"20"、"202"、"2023"的数量
    for (char ch: s) {
        if (ch == '2') {
            dp[0]++;
            dp[2] += dp[1];
        }
        if (ch == '0') dp[1] += dp[0];
        if (ch == '3') dp[3] += dp[2];
    }
    return dp[3];
}

int main() {
    for (ll i = 1; i <= 2023; i++) { // 剪枝
        s += to_string(i);
    }
    cout << f1() << endl; // 暴力
    cout << f2() << endl; // dp
    return 0;
}
```
