> 递推 + 高精度：f[i]=f[i-1]+f[i-2] 为斐波那契，每项用数组逐位存大数，加法后处理进位，最后倒序输出。

# 高精度 & 递推
``` c++
#include <iostream>

using namespace std;

int arr[5003][5003];

int main() {
    int n, len = 1; // 位数
    cin >> n;
    // 初始数据
    arr[1][1] = 1;
    arr[2][1] = 2;
    int i;
    for (i = 3; i <= n; ++i) {
        for (int j = 1; j <= len; ++j) {
            arr[i][j] = arr[i-1][j] + arr[i-2][j];
            
        }
        for (int j = 1; j <= len ; ++j) {
            if (arr[i][j] > 9) {
                arr[i][j+1]+=arr[i][j]/10;//向高一位进1
                arr[i][j]=arr[i][j]%10;
                if(arr[i][len+1]) len++;
            }
        }


    }
    if (n == 1) cout << 1;
    else {
        for (int j = len; j >= 1; --j) {
            cout << arr[i - 1][j];
        }
    }
    return 0;
}
```
