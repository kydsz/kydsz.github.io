> 完全背包：把每个平方数当作可重复取的物品，f[j][l] 表示和为 j 用 l 个平方数的方案数，累加 l=1..4 得答案。

## 背包dp





#### 就是一个背包，把每个平方数看成一个物品，求方案数。

```c++

#include<cstdio>
#include<iostream>

using namespace std;
int f[32770][5];

int main() {
    int i, j, l, n = 32768, t, ans;
    f[0][0] = 1;
    for (i = 1; i * i <= n; i++)
        for (j = i * i; j <= n; j++)
            for (l = 1; l <= 4; l++)
                f[j][l] += f[j - i * i][l - 1];
    scanf("%d", &t);
    while (t--) {
        ans = 0;
        scanf("%d", &n);
        for (i = 1; i <= 4; i++)
            ans += f[n][i];
        printf("%d\n", ans);
    }
    return 0;
}
```
