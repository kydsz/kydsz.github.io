> N 进制高精度加法：字符串模拟逐位相加并进位，每步将原串反转相加，判断是否在 30 步内成为回文。

## 进制	高精度

```c++
#include <bits/stdc++.h>

using namespace std;
int n, step = 0;
string s;

bool isHW(const string &x) {
    for (int i = 0, j = (int) x.length() - 1; i < x.length() && i != j; ++i, --j) {
        if (x[i] != x[j]) return false;
    }
    return true;
}

string sps(string a, string b) {
    string ans;
    int sum, i1, i2, p = 0;
    for (int i = a.length() - 1; i >= 0; --i) {
        // char to int
        if (a[i] >= '0' && a[i] <= '9') {
            i1 = a[i] - '0';
        } else {
            i1 = a[i] - 'A' + 10;
        }
        if (b[i] >= '0' && b[i] <= '9') {
            i2 = b[i] - '0';
        } else {
            i2 = b[i] - 'A' + 10;
        }

        sum = i1 + i2 + p;

        if (sum >= n) {
            p = sum / n;
            sum -= n;
        } else {
            p = 0;
        }
        if (n <= 10) {
            ans = char(sum + '0') + ans;
        } else if (sum > 9) {
            ans = (char) (sum - 10 + 'A') + ans;
        } else {
            ans = (char) (sum + '0') + ans;
        }

    }

    if (p) {
        if (p <= 10) {
            ans = char(p + '0') + ans;
        } else {
            ans = char(p + 'A' - 10) + ans;
        }
    }
    return ans;
}

bool add() {
    for (int i = 0; i <= 30; ++i) {
        if (isHW(s)) return true;
        /*
        string ts;
        for (char x: s) {
            ts = x + ts;
        }*/
        string ts = s;
        reverse(s.begin(), s.end()); // 反转字符串
        
        step++;
        s = sps(s, ts);
    }
    return false;
}

int main() {
    cin >> n;
    cin >> s;
    if (add()) {
        cout << "STEP=" << step;
    } else {
        cout << "Impossible!";
    }
    return 0;
}
```
