> 枚举区间内每个日期，判断年份反转后与月日相同即回文日期。

## 枚举

```c++
#include <iostream>

using namespace std;
bool year(int y) {
    if (y % 4 == 0) {
        if (y % 400 == 0) {
            return true;
        }
        if (y % 100 != 0) {
            return true;
        }
    }
    return false;
}
int to_num(char c) {
    switch (c) {
        case '1': return 1;
        case '2': return 2;
        case '3': return 3;
        case '4': return 4;
        case '5': return 5;
        case '6': return 6;
        case '7': return 7;
        case '8': return 8;
        case '9': return 9;
        default: return 0;
    }
}
int main() {
    string d1, d2;
    int ans = 0;
    int arr[12] = {31, 28,31, 30, 31,30, 31, 31,30, 31, 30,31};
    cin >> d1 >> d2;
    int sy = 0, sm = 0, sd = 0;
    int ey = 0, em = 0, ed = 0;

    for (int i = 0; i < 4; ++i) {
        sy = sy * 10 + to_num(d1[i]);
        ey = ey * 10 + to_num(d2[i]);
    }
    for (int i = 4; i < 6; ++i) {
        sm = sm * 10 + to_num(d1[i]);
        em = em * 10 + to_num(d2[i]);
    }
    for (int i = 6; i < 8; ++i) {
        sd = sd * 10 + to_num(d1[i]);
        ed = ed * 10 + to_num(d2[i]);
    }
    for (int i = sy; i <= ey; ++i) {
        if (year(i)) {
            arr[1] = 29;
        } else {
            arr[1] = 28;
        }
        int j = 1;
        if (i == sy) j = sm;
        while (j <= 12) {
            if (i == ey && j > em) break;
            int k = 1;
            if (i == sy && j == sm) k = sd;
            while (k <= arr[j - 1]) {
                if (i == ey && j == em && k > ed) break;
                string s1, s2;
                int y = i;

                while (y){
                    s1 = s1 + to_string( y % 10);
                    y /= 10;
                }
                if (j < 10) {
                    s2 = s2 + '0';

                }
                s2 = s2 + to_string(j);
                if (k < 10) {
                    s2 = s2 + '0';
                }
                s2 = s2 + to_string(k);
                if (s1 == s2) ans++;
                k++;
            }
            j++;
        }
    }
    cout << ans;
    return 0;
}
```
