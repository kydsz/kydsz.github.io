> 二分答案：二分相邻牛最小距离的最大值，check 从第一个隔间开始贪心放置，间距不小于 m 就放牛，能放够 c 头则说明可行。

## 二分

```c++
#include <iostream>
#include <algorithm>

using namespace std;
int n, c, a[100005];

bool check(int m)
{
    int cnt = 0;
    int d;
    int last = a[1]; // 上一个
    for (int i = 2; i <= n; i++) {
        d = a[i] - last;
        if (d >= m) {
            cnt++;
            last = a[i];
        }
    }
    if (cnt + 1 < c)return true;
    else return false;
}

int main() {
    cin >> n >> c;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    sort(a + 1, a + n + 1);
    int l = 0, r = a[n] - a[1], mid;
    while (l <= r) {
        mid = (r + l) >> 1;
        if (check(mid))
            r = mid - 1;
        else
            l = mid + 1;
    }
    cout << r << endl;
    return 0;
}
```
