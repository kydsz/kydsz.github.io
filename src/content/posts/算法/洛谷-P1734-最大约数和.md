> 01 背包：每个数 i 视为体积 i、价值为其约数和（不含自身）的物品，容量 s 内倒序转移 dp[j]=max(dp[j], dp[j-i]+约数和(i)) 求最大价值。

## dp动态规划	01背包	

```c++
#include <bits/stdc++.h>
using namespace std;
int s, dp[1010];

// 求
int ff(int x) {
    int ans = 0;
    for (int i = 1; i < x; ++i) {
        if (x % i == 0) {
            ans += i;
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin >> s;
    for (int i = 1; i <= s; ++i) {
        for (int j = s; j >= i; --j) {
            dp[j] = max(dp[j], dp[j - i] + ff(i));
        }
    }
    cout << dp[s];
    return 0;
}
```
