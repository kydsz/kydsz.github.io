> 前缀和统计，枚举平方数 j²，存在前缀和之差等于 j² 即构成合法区间。

# 牛客 题目：珂朵莉与宇宙
## 前缀和算法
```c++
#include<iostream>
#define SIZE 1000010
using namespace std;
int arr[SIZE];
int S[SIZE] = {1};
int main(){
    int n, res = 0;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> arr[i];
        arr[i] = arr[i-1] + arr[i];
    }
//    arr为前缀和数组
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j <= SIZE; ++j) {
            if (arr[i]-j*j >= 0) res += S[arr[i]-j*j];
            else break;
        }
        S[arr[i]]++;
    }
    cout << res;
    return 0;
}
```
