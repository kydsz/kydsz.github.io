> 从最大可能公差向下枚举模拟补齐，取能覆盖全部项的最短长度。

## 暴力90%

```c++
#include<iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
int n, ans = 0x7fffffff;
ll arr[100010];
int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> arr[i];
    }
    sort(arr, arr + n);

    for (ll i = arr[1] - arr[0]; i >= 1; --i) {
        int len = 1;
        bool  p = true;
        ll t = arr[0];
        ll index = 1;
        while (true) {
            t += i;
            if (index == n) {
                break;
            } else if (t > arr[index]) {
                p = false;
                break;
            } else if (t == arr[index]) {
                index++;
            }
            len++;
        }
        if (p) {
            ans = min(ans, len);
        }
    }
    cout << ans;
    return 0;
}
```
