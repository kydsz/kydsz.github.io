> 前缀和：统计模 k 同余的前缀和个数，两两组合即 k 倍区间。

# 前缀和

``` c++
#include <iostream>

using namespace std;
const int MAX = 200000;
long n, k, ans = 0;
long arr[MAX];
long S[MAX];

int main() {
    cin >> n >> k;
    S[0] = 1;
    for (int i = 1; i <= n; ++i) {
        long a;
        cin >> a;
        arr[i] = arr[i - 1] + a;
        S[arr[i] % k]++ ;
    }
    for (int i = 0; i < k; ++i) {
        ans += S[i] * (S[i] - 1) / 2;
    }
    cout << ans;
    return 0;
}
```
