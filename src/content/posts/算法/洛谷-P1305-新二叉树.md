> 二叉树先序遍历：用数组记录每个节点的左右孩子下标，从根递归按“根左右”顺序输出即可。

# 二叉树

```c++
#include <iostream>
using namespace std;

struct Node{
    char data;
    int L = -1, R = -1;
}tree[100];

void DLR(int x){
    cout << tree[x].data;
    // 判断左右孩子是否存在，需要分开判断
    if (tree[x].L != -1) DLR(tree[x].L);
    if (tree[x].R != -1) DLR(tree[x].R);
}


int main(){
    int n, index = -1;
    char root, d_l, d_r;
    cin >> n;
    while (n--){
        cin >> root >> d_l >> d_r;
        if (index == -1) // 初始化根节点
            index = root-'a';
        tree[root-'a'].data = root;
        if (d_l != '*') tree[root-'a'].L = d_l-'a';
        if (d_r != '*') tree[root-'a'].R = d_r-'a';
    }
    DLR(index);
    return 0;
}
```
