> 马的遍历：从起点 BFS，按马走日的 8 个方向扩展，首次到达即为最短步数，逐层填充整张棋盘。

# bfs

```c++
#include<bits/stdc++.h>

#define LEN 1000
using namespace std;
int n, m;
int step[LEN][LEN], state[LEN][LEN];
struct _map {
    int x, y;
} map[LEN];
queue<_map> q;


int dx[8] = {1, 1, -1, -1, 2, 2, -2, -2};
int dy[8] = {2, -2, 2, -2, 1, -1, 1, -1};

void bfs(int x, int y) {
    step[x][y] = 0;
    q.push({x, y});
    state[x][y] = 1;
    while (!q.empty()) {
        _map f = q.front();
        q.pop();
        for (int i = 0; i < 8; ++i) {
            int tx = f.x + dx[i];
            int ty = f.y + dy[i];
            if (tx < 1 || ty < 1 || tx > n || ty > m || state[tx][ty] == 1) continue;
            state[tx][ty] = 1;
            step[tx][ty] = step[f.x][f.y] + 1;
            q.push({tx, ty});
        }
    }
}

int main() {
    int x, y;
    cin >> n >> m >> x >> y;
    memset(step, -1, sizeof(step));
    bfs(x, y);
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            cout << step[i][j] << "    ";
        }
        cout << endl;
    }
    return 0;
}
```
