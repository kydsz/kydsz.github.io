> DFS 回溯按位选未用过的数生成全排列，另附 next_permutation 解法。

# dfs解法
```c++
#include<iostream>
using namespace std;

/*
 * arr 待排列的数组
 * state 代表状态数组值
 * len 代表我们当前需要排序数组的长度
 * index 代表我们当前递归层次选择的是第几个元素
 * res 代表寄存当前选择值的数组
 * */

// 打印函数应该打印当前排列结果 res 而不是原数组 arr
void print(int res[], int len){
    for (int i = 0; i < len; ++i) {
        cout << "    " << res[i]; // 注意输出格式，根据实际情况修改
    }
    cout << endl;
}

// 生成全排列
void all_perm(int arr[], int state[], int len, int index, int res[]){
    if (index == len){
        print(res, len); // 打印当前排列
        return;
    }
    for (int i = 0; i < len; ++i) {
        if (state[i] == 0){ // 如果这个数还没有被选择
            state[i] = 1; // 标记为已选择
            res[index] = arr[i]; // 记录当前选择的元素
            all_perm(arr, state, len, index + 1, res); // 递归
            state[i] = 0; // 回溯，撤销选择
        }
    }
}

// 主函数
int main(){
    int n, arr[10];
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        arr[i-1] = i; // 初始化待排列的数组（1 到 n）
    }
    int state[10] = {0}; // 状态数组初始化为 0
    int res[10]; // 用来存放当前的排列组合
    all_perm(arr, state, n, 0, res); // 调用函数生成全排列
    return 0;
}
```

# 全排函数next_permutation
```c++
#include<iostream>
#include<algorithm>
using namespace std;

int main()
{
    int n,a[1000];
    cin >> n;
    for(int i = 0 ; i < n ; i++)
        a[i] = i+1;

    do{
        for(int i = 0 ; i < n ; i++)
            cout << a[i] << " ";
        cout << endl;
    }while(next_permutation(a, a + n));
    return 0;
}
```
