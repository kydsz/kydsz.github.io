> 按单位价值 v/m 降序排序贪心装入，不足容量时按比例取部分。

# P2240 【深基12.例1】部分背包问题

```c++
#include <iostream>
#include<algorithm>
using namespace std;
struct node {
    int m, v;
    double j;
}a[110];
bool cmp(node x, node y) {
    return x.j > y.j;
}
int main()
{
    int n, t;
    cin >> n >> t;
    for (int i = 0; i < n; i++) {
        cin >> a[i].m >> a[i].v;
        a[i].j = (double)a[i].v /(double) a[i].m;
    }
    sort(a, a + n, cmp);
    int sum = 0;
    double ans = 0;
    for (int i = 0; i < n; i++) {
        if (sum <t) {
            if (a[i].m <= (t - sum)) {
                sum += a[i].m;
                ans += a[i].v;
            }
            else {
                ans += (t - sum) * a[i].j;
                break;
            }
        }
        else {
            break;
        }
    }
    printf("%.2lf", ans);

    return 0;
}
```
