> 贪心：按接水时间升序排序，耗时短者先接水使总等待最小，第 i 人耗时累乘其后剩余人数即为总等待时间。

# 贪心  & 排序
```c++
#include <iostream>
#include <algorithm>

using namespace std;

struct _water {
    // t表示时间 num表示序数
    double t;
    int num;
}water[1010];

bool comp(struct _water a, struct _water b){
    return a.t < b.t;
}

int main() {
    int n;
    double time = 0;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> water[i].t;
        water[i].num = i;
    }

    sort(water+1,water+n+1, comp);

    for (int i = 1; i <= n; ++i) {
        cout << water[i].num << " ";
    }
    cout << endl;

    for (int i = 1; i < n; ++i) {
        int j = n - i;
        time += water[i].t * j;
        /*10
         * 56 12 1 99 1000 234 33 55 99 812
         * 以这个输入为例子当第一个人打水是1分钟，剩下9人都要等待1分钟所以time = 1 * 9
         */
    }
    printf("%.2lf", time / n);
    return 0;
}
```
