> 快速幂：把指数 n 按二进制拆解，边平方边累乘，O(log n)；第二份是带取模（mol）的版本，常用于求大数幂的模。

```c++
long long quickPow(long x, long n){
	long long ans = 1;
    while(n){
        if(n&1){
            ans *= x;
        }
        x *= x;
        n >>= 1;
    }
    return ans;
}

long long quickPow(long long x, long long n, long long mol){
	long long ans = 1;
    while(n){
        if(n&1){
            ans = (ans * x) % mol;
        }
        x = (x * x) % mol;
        n >>= 1;
    }
    return ans;
}
```
